Chapter 01

Introduction to C Language

C एउटा general-purpose, procedural programming language हो जुन 1972 मा Dennis Ritchie ले Bell Labs मा बनाएका थिए।

📌 C Language किन महत्त्वपूर्ण छ?

  • सबैभन्दा पुरानो र आधारभूत language — सबै language को जड
  • Operating Systems (Linux, Windows) C मा लेखिएका छन्
  • Fast र efficient — hardware नजिक काम गर्छ
  • BIT को foundation subject — यो बुझ्यो भने बाकी सब सजिलो

📝 C Program को Basic Structure

hello.c 
#include <stdio.h>    // Header file — input/output ko lagi

int main() {          // Main function — program yahi bata suru
    printf("Namaste Duniya!\n");  // Output print garne
    return 0;           // 0 = success
}
🔴 Exam Important #include <stdio.h> बिना printf() र scanf() काम गर्दैन। हरेक C program मा यो हुनुपर्छ।

⚙️ C Program कसरी Run हुन्छ?

Source Code (.c) → Compiler (gcc) → Object Code → Linker → Executable (.exe)

💡 याद गर्नु Compilation steps: Preprocessing → Compilation → Assembly → Linking

Chapter 05

Control Flow — if/else, switch

Conditions हेरेर program को direction तय गर्ने।

🔀 if-else Statement

iffelse.c
int marks = 75;

if (marks >= 80) {
    printf("Distinction!\n");
} else if (marks >= 60) {
    printf("First Division\n");
} else if (marks >= 45) {
    printf("Second Division\n");
} else {
    printf("Fail!\n");
}

🔃 switch Statement

switch.c
int day = 3;
switch(day) {
    case 1: printf("Aaitabar"); break;
    case 2: printf("Sombar");  break;
    case 3: printf("Mangalbar"); break;
    default: printf("Invalid");
}
⚠️ break बिर्सनु नहोस्! break नभएमा सबै case execute हुन्छन् — यसलाई "fall-through" भनिन्छ।

❓ Ternary Operator

ternary.c
// condition ? value_if_true : value_if_false
int a = 10, b = 20;
int max = (a > b) ? a : b;   // max = 20
printf("%s", (a%2==0) ? "Even" : "Odd");
Chapter 06

Loops — for, while, do-while

Code लाई repeatedly execute गर्ने mechanism।

🔁 for Loop

for_loop.c
// 1 देखि 10 सम्म print गर्ने
for(int i=1; i<=10; i++) {
    printf("%d ", i);
}

// Factorial
int n=5, fact=1;
for(int i=1; i<=n; i++) {
    fact *= i;
}
printf("Factorial: %d", fact);  // 120

🔄 while Loop

while.c
int i = 1;
while(i <= 5) {
    printf("%d\n", i);
    i++;    // Yo bhulyo bhaney infinite loop!
}

🔃 do-while Loop

dowhile.c
int i = 1;
do {
    printf("%d\n", i);
    i++;
} while(i <= 5);
// Farak: Condition false bhaye pani EKPALTA execute huncha!
💡 Kura Yaad Garnu for = count थाहा छ | while = condition थाहा छ | do-while = कम्तिमा एकपटक run गर्नुपर्छ

Chapter 07

Functions

Code लाई reusable blocks मा organize गर्ने तरिका।

🧩 Function Syntax

functions.c
#include <stdio.h>

// Function Declaration (Prototype)
int add(int a, int b);

// Function Definition
int add(int a, int b) {
    return a + b;
}

int main() {
    int result = add(5, 3);   // Function Call
    printf("Sum = %d", result);  // 8
    return 0;
}

🔂 Recursion

recursion.c
// Recursive Factorial
int factorial(int n) {
    if(n == 0 || n == 1)
        return 1;           // Base case — must have!
    return n * factorial(n-1);  // Recursive call
}
// factorial(5) = 5*4*3*2*1 = 120
🔴 Exam Must Know Recursion को base case बिर्सनु भएन — नत्र infinite loop आउँछ।

📮 Call by Value vs Call by Reference

callby.c
// Call by Value — original change hudaina
void byValue(int x) { x = 100; }

// Call by Reference — original change huncha
void byRef(int *x) { *x = 100; }

int main() {
    int a = 5;
    byValue(a);     // a still 5
    byRef(&a);      // a is now 100
}
Chapter 08

Arrays

Same data type को multiple values एउटै variable मा store गर्ने।

📦 1D Array

array.c
int marks[5] = {90, 85, 78, 92, 88};

// Access: index 0 bata suru
printf("%d", marks[0]);   // 90
printf("%d", marks[4]);   // 88

// Loop bata print garne
for(int i=0; i<5; i++) {
    printf("%d ", marks[i]);
}
🔴 याद गर्नु! Array index 0 बाट सुरु हुन्छ! 5 elements भएको array: marks[0] देखि marks[4] सम्म।

🔲 2D Array (Matrix)

matrix.c
int matrix[3][3] = {
    {1, 2, 3},
    {4, 5, 6},
    {7, 8, 9}
};

// matrix[row][column]
printf("%d", matrix[1][2]);  // 6

for(int i=0; i<3; i++) {
    for(int j=0; j<3; j++) {
        printf("%d ", matrix[i][j]);
    }
    printf("\n");
}

Chapter 10

Pointers

Pointer = variable को memory address store गर्ने variable। C को सबैभन्दा महत्त्वपूर्ण concept!

📍 Pointer Basics

pointers.c
int num = 10;
int *ptr;          // Pointer declaration

ptr = &num;        // & = address-of operator

printf("%d", num);   // 10 (value)
printf("%p", ptr);   // Address (eg: 0x7fff...)
printf("%d", *ptr);  // 10 (* = dereference = value at address)

*ptr = 20;          // num ko value change huncha!
printf("%d", num);   // 20
🔴 Key Points & = address linu | * = value herna (dereference) | Pointer le address store garcha

🔗 Pointer & Array

ptr_array.c
int arr[] = {10, 20, 30};
int *p = arr;     // arr ko first element ko address

printf("%d", *p);       // 10
printf("%d", *(p+1));   // 20
printf("%d", *(p+2));   // 30
p++;                      // Next element ma jaancha
Chapter 11

Structures

Different data types लाई एकसाथ राख्ने user-defined data type।

🏗️ Structure Definition & Use

struct.c
#include <stdio.h>

// Structure define garne
struct Student {
    char name[50];
    int roll;
    float marks;
};

int main() {
    struct Student s1;    // Variable banauney

    s1.roll = 101;       // . (dot) operator bata access
    s1.marks = 85.5;

    printf("Roll: %d, Marks: %.1f", s1.roll, s1.marks);
    return 0;
}

🔑 typedef & Struct Pointer

typedef.c
// typedef bata shortcut
typedef struct {
    int x, y;
} Point;

Point p1 = {3, 4};     // "struct" likhnu pardaina

// Pointer bata access: -> operator
Point *ptr = &p1;
printf("%d", ptr->x);    // 3 (dot ko saatta arrow)

File Handling

Program बाट files read/write गर्ने।

📁 File Operations

ModeMeaning
"r"Read — file parhna
"w"Write — file ma lekhna (overwrite)
"a"Append — file ko antima add garne
"r+"Read + Write
"w+"Write + Read (overwrite)

💾 File Read & Write

fileio.c
#include <stdio.h>

int main() {
    FILE *fp;                   // File pointer

    // Write to file
    fp = fopen("data.txt", "w");
    fprintf(fp, "Namaste BIT!\n");
    fclose(fp);                 // File close garna na birse!

    // Read from file
    char buffer[100];
    fp = fopen("data.txt", "r");
    while(fgets(buffer, 100, fp) != NULL) {
        printf("%s", buffer);
    }
    fclose(fp);

    return 0;
}
🔴 Exam याद गर्नु fopen() fail भएमा NULL return गर्छ। fclose() नगरे data lose हुन सक्छ!
